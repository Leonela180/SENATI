== images
  id                     bigint                     NOT NULL  nextval('images_id_seq'::regclass)
  owner_user_id          bigint                     NOT NULL  
  workspace_id           bigint                     NOT NULL  
  original_storage_key   text                       NOT NULL  
  original_filename      character varying          NOT NULL  
  mime_type              character varying          NOT NULL  
  byte_size              bigint                     NOT NULL  
  width                  integer                    NOT NULL  
  height                 integer                    NOT NULL  
  source_type            character varying          NOT NULL  'ai_generated'::character varying
  visibility             character varying          NOT NULL  'private'::character varying
  created_at             timestamp with time zone   NOT NULL  
  [ck_images_byte_size] CHECK ((byte_size > 0))
  [ck_images_dimensions] CHECK (((width > 0) AND (height > 0)))
  [ck_images_visibility] CHECK (((visibility)::text = ANY ((ARRAY['private'::character varying, 'team'::character varying])::text[])))
  [images_original_storage_key_key] UNIQUE (original_storage_key)
  [images_owner_user_id_fkey] FOREIGN KEY (owner_user_id) REFERENCES users(id)
  [images_pkey] PRIMARY KEY (id)
  [images_workspace_id_fkey] FOREIGN KEY (workspace_id) REFERENCES workspaces(id)

  [ck_jobs_completed_result] CHECK ((((status)::text <> 'completed'::text) OR (result_storage_key IS NOT NULL)))
  [ck_jobs_status] CHECK (((status)::text = ANY ((ARRAY['queued'::character varying, 'processing'::character varying, 'completed'::character varying, 'failed'::character varying])::text[])))
  [processing_jobs_image_id_fkey] FOREIGN KEY (image_id) REFERENCES images(id)
